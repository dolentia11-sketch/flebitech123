def calcular_cotizacion_determinista(afiliado, producto_id, plan_nivel):
    # 1. Base tarifaria MetLife según convenio Colsubsidio
    TARIFAS_BASE = {
        'vida_voluntaria': 18500,
        'accidentes_personales': 12000,
        'mascotas_salud': 15000,
        'desempleo_protegido': 22000
    }
    
    # 2. Factor por Categoría de Afiliación (Subsidio)
    DESCUENTOS = {'Cat A': 0.35, 'Cat B': 0.20, 'Cat C': 0.0, 'No Afiliado': -0.10}
    
    # 3. Factor Cobertura (Plan Básico x1, Estándar x1.4, Premium x1.8)
    MULT_PLAN = {'basico': 1.0, 'estandar': 1.4, 'premium': 1.8}
    
    base = TARIFAS_BASE.get(producto_id, 15000)
    factor_cat = 1.0 - DESCUENTOS.get(afiliado.categoria, 0.0)
    mult = MULT_PLAN.get(plan_nivel, 1.0)
    
    # Cálculo strictly determinista
    prima_mensual = round(base * factor_cat * mult, -2)
    cobertura_total = int(prima_mensual * 1200) # Factor actuarial
    
    return {
        "prima_mensual": prima_mensual,
        "cobertura_monto": cobertura_total,
        "hash_idoneidad": hashlib.sha256(f"{afiliado.id}:{producto_id}:{prima_mensual}".encode()).hexdigest()[:16]
    }
